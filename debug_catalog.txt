  const generateCatalog = () => {
    const win = window.open('', '_blank');
    if (!win) return;
    const inStockVehicles = vehicles.filter(v => v.stock > 0);
    const cardsPerPage = 4;
    const totalPages = Math.ceil(inStockVehicles.length / cardsPerPage);
    const dateStr = new Date().toLocaleDateString('en-GB', { day: 'numeric', month: 'long', year: 'numeric' });

    let pagesHtml = '';
    for (let i = 0; i < inStockVehicles.length; i += cardsPerPage) {
      const chunk = inStockVehicles.slice(i, i + cardsPerPage);
      const pageNum = Math.floor(i / cardsPerPage) + 1;
      const isFirstPage = pageNum === 1;

      const cardsHtml = chunk.map(v => {
        const imgSrc = v.image_url ? `http://localhost:5001${v.image_url}` : '';
        return `
          <div class="card">
            <div class="card-img">
              ${imgSrc ? `<img src="${imgSrc}" alt="${v.brand}" />` : '<div class="no-img">No Image</div>'}
            </div>
            <div class="card-body">
              <div class="card-header">
                <h2>${v.brand}</h2>
                <span class="price">Rs. ${Number(v.price).toLocaleString()}</span>
              </div>
              <div class="tags">
                ${v.category ? `<span class="tag">${v.category}</span>` : ''}
                <span class="tag fuel">${v.fuel_type || 'Petrol'}</span>
                <span class="tag stock">${v.stock} in stock</span>
              </div>
              ${v.description ? `<p class="desc">${v.description}</p>` : ''}
            </div>
          </div>`;
      }).join('');

      pagesHtml += `
        <div class="page">
          ${isFirstPage ? `
            <div class="header">
              <div>
                <h1>Mohan Trading</h1>
                <p style="font-size:12px;color:#666;margin-top:3px;">Available Vehicle Catalog</p>
              </div>
              <div class="meta">
                <strong>${inStockVehicles.length} vehicles available</strong>
                ${dateStr}
              </div>
            </div>
          ` : `
            <div class="page-header-compact">
               <span class="brand-line">Mohan Trading Catalog</span>
               <span class="pg-num">Page ${pageNum} of ${totalPages}</span>
            </div>
          `}
          <div class="grid">${cardsHtml}</div>
        </div>
      `;
    }


    win.document.write(`<!DOCTYPE html>
<html>
<head>
  <meta charset="UTF-8" />
  <title>Mohan Trading — Vehicle Catalog</title>
  <style>
    @page { size: A4 portrait; margin: 0; }
    * { margin: 0; padding: 0; box-sizing: border-box; }
    html, body { width: 210mm; background: white; font-family: 'Segoe UI', Arial, sans-serif; color: #1a1a2e; }

    /* ── Watermark ── */
    body::before {
      content: 'MOHAN TRADING';
      position: fixed;
      top: 50%;
      left: 50%;
      transform: translate(-50%, -50%) rotate(-35deg);
      font-size: 62px;
      font-weight: 900;
      letter-spacing: 6px;
      color: rgba(26, 26, 46, 0.055);
      white-space: nowrap;
      pointer-events: none;
      z-index: 0;
    }

    .page { width: 210mm; height: 297mm; padding: 15mm 15mm; page-break-after: always; position: relative; z-index: 1; overflow: hidden; display: flex; flex-direction: column; }
    .page:last-child { page-break-after: auto; }
    .header { display: flex; justify-content: space-between; align-items: flex-end; border-bottom: 3px solid #1a1a2e; padding-bottom: 12px; margin-bottom: 20px; }
